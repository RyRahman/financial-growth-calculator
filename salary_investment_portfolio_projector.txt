        function runSalaryCalculations() {
            const startSalary = parseFloat(startingSalaryInput.value) || 0;
            const raisePct = (parseFloat(annualRaiseInput.value) || 0) / 100;
            const taxPct = (parseFloat(taxRateInput.value) || 0) / 100;
            const inflationPct = (parseFloat(inflationRateInput.value) || 0) / 100;
            const years = parseInt(salaryCustomYearsInput.value) || 30;

            salaryCalculatedData = [];
            let currentGross = startSalary;
            let cumulativeNet = 0;
            let cumulativeTaxes = 0;

            for (let year = 1; year <= Math.max(years, 40); year++) {
                const taxes = currentGross * taxPct;
                const netAnnual = currentGross - taxes;
                const netMonthly = netAnnual / 12;
                cumulativeNet += netAnnual;
                cumulativeTaxes += taxes;

                const inflationFactor = Math.pow(1 + inflationPct, year - 1);
                const realNetAnnual = netAnnual / inflationFactor;

                salaryCalculatedData.push({
                    year: year,
                    grossAnnual: currentGross,
                    taxes: taxes,
                    netAnnual: netAnnual,
                    netMonthly: netMonthly,
                    realNetAnnual: realNetAnnual,
                    cumulativeNet: cumulativeNet,
                    cumulativeTaxes: cumulativeTaxes
                });

                currentGross = currentGross * (1 + raisePct);
            }

            updateSalaryUI(years);
        }

        function runInvestmentCalculations() {
            const initial = parseFloat(initialInvestmentInput.value) || 0;
            const contribution = parseFloat(contributionAmountInput.value) || 0;
            const annualReturnPct = (parseFloat(investmentReturnInput.value) || 0) / 100;
            const gainsTaxPct = (parseFloat(gainsTaxRateInput.value) || 0) / 100;
            const years = parseInt(investCustomYearsInput.value) || 30;

            let periodsPerYear = 12;
            let freqLabel = '/mo';
            if (contributionFrequency === 'weekly') {
                periodsPerYear = 52;
                freqLabel = '/wk';
            } else if (contributionFrequency === 'annual') {
                periodsPerYear = 1;
                freqLabel = '/yr';
            }

            const periodRate = annualReturnPct / periodsPerYear;
            const annualDeposit = contribution * periodsPerYear;

            document.getElementById('currentValDisplay').textContent = formatCurrency(initial);
            document.getElementById('contributionSummaryNote').textContent = `($${contribution}${freqLabel} = ~$${annualDeposit.toLocaleString()}/year)`;

            investmentCalculatedData = [];
            let currentBalance = initial;
            let totalPrincipal = initial;

            for (let y = 1; y <= Math.max(years, 40); y++) {
                const startBal = currentBalance;
                let yearGrowth = 0;

                for (let p = 0; p < periodsPerYear; p++) {
                    currentBalance += contribution;
                    totalPrincipal += contribution;
                    const periodInterest = currentBalance * periodRate;
                    yearGrowth += periodInterest;
                    currentBalance += periodInterest;
                }

                const totalInterestEarned = currentBalance - totalPrincipal;
                const taxOnGains = Math.max(0, totalInterestEarned * gainsTaxPct);
                const netFutureValue = currentBalance - taxOnGains;

                investmentCalculatedData.push({
                    year: y,
                    startingBalance: startBal,
                    annualDeposit: annualDeposit,
                    growthEarned: yearGrowth,
                    totalPrincipal: totalPrincipal,
                    endingBalance: currentBalance,
                    totalInterest: totalInterestEarned,
                    taxOnGains: taxOnGains,
                    netFutureValue: netFutureValue
                });
            }

            updateInvestmentUI(years);
        }
